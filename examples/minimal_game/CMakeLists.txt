# ==========================================================================
# phyxel_minimal_game — Example standalone game using EngineRuntime
# ==========================================================================
# Demonstrates the simplest way to build a game against phyxel_core
# without the Application monolith.

add_executable(phyxel_minimal_game
    main.cpp
    MinimalGame.cpp
    MinimalGame.h
)

# Link directly against phyxel_core — no need for phyxel_editor
target_link_libraries(phyxel_minimal_game PRIVATE phyxel_core)

# Include path for local headers
target_include_directories(phyxel_minimal_game PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}
)

# A standalone example must be runnable from its build output, not merely
# compile. Keep this in lockstep with the asset layout generated by
# tools/create_project.py so CI exercises the same boundary games use.
add_custom_command(TARGET phyxel_minimal_game POST_BUILD
    COMMENT "Copying minimal-game runtime assets"
    COMMAND ${CMAKE_COMMAND} -E copy_directory
        "${PHYXEL_ROOT_DIR}/shaders"
        "$<TARGET_FILE_DIR:phyxel_minimal_game>/shaders"
    COMMAND ${CMAKE_COMMAND} -E copy_directory
        "${PHYXEL_ROOT_DIR}/resources"
        "$<TARGET_FILE_DIR:phyxel_minimal_game>/resources"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${CMAKE_CURRENT_SOURCE_DIR}/engine.json"
        "$<TARGET_FILE_DIR:phyxel_minimal_game>/engine.json"
)

if(MSVC)
    set_property(TARGET phyxel_minimal_game PROPERTY
        MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    set_target_properties(phyxel_minimal_game PROPERTIES
        LINK_FLAGS "/NODEFAULTLIB:python313t.lib")
endif()
